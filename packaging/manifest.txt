# ВАЖНО: сюда попадают только файлы, которые есть в git. Приватные рабочие

#   init  — init-скрипты Entware        -> /opt/etc/init.d/     (в архиве: корень)
#   doc   — документация                -> /opt/susanin/docs/   (в архиве: docs/)

#   cfg   — пример конфига              -> share/susanin.conf.example
#   www   — статика панели (каталог)    -> /opt/susanin/www/    (в архиве: www/)
#   root  — файлы только для tar-архива (в .ipk не нужны)
